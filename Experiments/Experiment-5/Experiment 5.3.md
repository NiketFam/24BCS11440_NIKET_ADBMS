DO $$

DECLARE

AGE INT:=18;

BEGIN

IF AGE>=18 THEN 

RAISE NOTICE 'YOUT AGE IS % AND YOUT ARE ELIGIBLE TO VOTE', AGE;



ELSE

RAISE NOTICE 'YOUT AGE IS % AND YOUT ARE NOT  ELIGIBLE TO VOTE', AGE;



END IF;

END;

$$





DO $$

DECLARE

VAL INT:=17;

BEGIN

IF VAL>=8 AND VAL <=10 THEN 

RAISE NOTICE 'RANGE IS BETWEEN 1 TO 10';



ELSEIF VAL>10 AND VAL<=20 THEN

RAISE NOTICE 'RANGE IS BETWEEN 11 TO 20';



ELSEIF VAL>20 AND VAL<=30 THEN

RAISE NOTICE 'RANGE IS BETWEEN 21 TO 30';



ELSE 

RAISE NOTICE 'VALUE IS GREATER THAN  30';

END IF;

END;

$$
